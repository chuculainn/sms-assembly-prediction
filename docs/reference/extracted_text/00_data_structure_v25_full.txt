    sms_pkg = build_sms_transfer_package(sms_update, comp)
    gap = build_gap_field(sms_pkg, ContactDomain, SMSPointCloudMapping)
    previous_pkg = initialize_stage_package(gap)
    for stage in stages:
        K_Gamma = load_condensed_operator(I_red[stage], material_boundary_hash)
        F_Gamma = build_reduced_force_vector(I_stage[stage], actual_loads)
        if overconstraint_enabled(stage):
            ext = build_extended_gap_and_compliance(gap, K_Gamma, F_Gamma, LocatorGroups[stage])
            model = build_overconstraint_contact_model(ext)
            state = solve_extended_complementarity(model, warm_start=previous_pkg.mode)
        else:
            pred = stage_predict(previous_pkg, I_red[stage], I_stage[stage], ParameterLibrary)
            state = solve_complementarity(pred, warm_start=previous_pkg.mode)
        trace = build_contact_computation_trace(gap, state.q, state.W, dependencies)
        trace.attach_solution(state)
        if process_measurement_available(stage):
            decision = diagnose_update_target(state, process_meas[stage])
            state = constrained_update_and_resolve(state, decision, process_meas[stage])
        increment = build_stage_increment(previous_pkg, state)
        ledger.register(increment)
        previous_pkg = build_stage_posterior_package(state, increment)
    check_no_double_count(ledger)
    eta = condense_interface_error(stage_packages)
    pred = predict_KCP(sms_pkg, eta, I_pred, ledger)

14.4 防重复计算的实现约束
     每次 run_single_sample_fast_model 必须创建独立 ContributionLedger。
     阶段更新不得修改 SMSUpdateTransferPackage。
     任何 q/W/K/模式变化必须由 ReSolveRequirement 驱动缓存失效和重求。
     N-2-1 扩展对象的 locator/joint/clamp 源必须进入同一 source ledger，避免同一定位器变形既作为柔度又作为阶段贡
      献重复计入。
